const DEFAULT_COUNTRY = "Unknown";
const DEFAULT_BROWSER = "Unknown";
const DEFAULT_DEVICE = "desktop";
const DEFAULT_REFERRER = "Direct";
const DEFAULT_IP = "unknown";

const normalizeBucketInterval = (value) => {
  const normalized = String(value || "").trim().toLowerCase();
  return ["1 day", "1 hour"].includes(normalized) ? normalized : "1 day";
};

const normalizeMetricRows = (rows = [], fallback = DEFAULT_COUNTRY) => rows.map((row) => ({
  label: String(row.label || "").trim() || fallback,
  count: Number(row.count || 0)
}));

const getBucketExpression = () => {
  const interval = normalizeBucketInterval(env.analyticsQueryBucket);

  return {
    $dateTrunc: {
      date: "$clickedAt",
      unit: interval === "1 hour" ? "hour" : "day",
      timezone: "UTC"
    }
  };
};

const getBreakdownField = (field) => {
  if (field === "deviceType") {
    return "$deviceType";
  }

  return `$${field}`;
};

const aggregateBreakdown = async ({ code, field, limit, fallback }) => {
  const result = await ClickEvent.aggregate([
    {
      $match: {
        code
      }
    },
    {
      $group: {
        _id: getBreakdownField(field),
        count: { $sum: 1 }
      }
    },
    {
      $project: {
        _id: 0,
        label: "$_id",
        count: 1
      }
    },
    {
      $sort: {
        count: -1,
        label: 1
      }
    },
    {
      $limit: limit
    }
  ]);

  return normalizeMetricRows(result, fallback);
};

const aggregateChartData = async ({ code, days = 30 }) => {
  const lookbackDays = Math.max(1, Number(days) || 30);
  const startDate = new Date();

  startDate.setUTCDate(startDate.getUTCDate() - (lookbackDays - 1));
  startDate.setUTCHours(0, 0, 0, 0);

  const result = await ClickEvent.aggregate([
    {
      $match: {
        code,
        clickedAt: {
          $gte: startDate
        }
      }
    },
    {
      $group: {
        _id: getBucketExpression(),
        clicks: { $sum: 1 }
      }
    },
    {
      $project: {
        _id: 0,
        date: "$_id",
        clicks: 1
      }
    },
    {
      $sort: {
        date: 1
      }
    }
  ]);

  return result.map((row) => ({
    date: row.date,
    clicks: Number(row.clicks || 0)
  }));
};

const fetchRecentClickDocs = async (code, limit = 12) => ClickEvent.find({ code })
  .sort({ clickedAt: -1 })
  .limit(Math.max(1, Number(limit) || 12))

export const getLinkAnalytics = async (code, userId) => {
  const link = await Url.findOne({ code, userId }).lean();

  if (!link) {
    throw new AppError("Link not found.", 404);
  }

  const [chartData, countries, devices, referrers, browsers, recentClicks] = await Promise.all([
    aggregateChartData({ code }),
    aggregateBreakdown({ code, field: "country", limit: 5, fallback: DEFAULT_COUNTRY }),
    aggregateBreakdown({ code, field: "deviceType", limit: 5, fallback: DEFAULT_DEVICE }),
    aggregateBreakdown({ code, field: "referrer", limit: 5, fallback: DEFAULT_REFERRER }),
    aggregateBreakdown({ code, field: "browser", limit: 5, fallback: DEFAULT_BROWSER }),
    fetchRecentClickDocs(code)
  ]);

  return {
    link,
    overview: {
      totalClicks: Number(link.clickCount || 0),
      lastClickedAt: link.lastClickedAt || null
    },
    chartData,
    countries,
    devices,
    referrers,
    browsers,
    recentClicks: recentClicks.map((click) => ({
      id: click.eventId || String(click._id),
      occurredAt: click.clickedAt,
      referrer: click.referrer || DEFAULT_REFERRER,
      browser: click.browser || DEFAULT_BROWSER,
      deviceType: click.deviceType || DEFAULT_DEVICE,
      country: click.country || DEFAULT_COUNTRY,
      clientIp: click.clientIp || DEFAULT_IP
    }))
  };
};

const buildLegacyEventId = (payload) => createHash("sha1")
  .update(
    [

      payload.referrer || "",
      payload.browser || "",
      payload.deviceType || "",
      payload.country || "",

const resolveAnalyticsContext = async (payload, eventId) => ({
  occurredAt: payload.occurredAt ? new Date(payload.occurredAt) : new Date(),
  referrer: payload.referrer || "Direct",
  browser: payload.browser || "Unknown",
  deviceType: payload.deviceType || "desktop",

    country: payload.country,
    clientIp: payload.clientIp,

});

export const processClickEvent = async (payload) => {
  const eventId = getEventId(payload);
  const requestLogger = analyticsLogger.child({
    eventId,
    code: payload.code,
    requestId: payload.requestId
  });

  requestLogger.debug("analytics.process.started");

  try {
    const context = await resolveAnalyticsContext(payload, eventId);
    const urlRecord = await Url.findOne({ code: payload.code }).select("_id code userId");

    if (!urlRecord) {
      requestLogger.warn("analytics.process.url_missing");
      return;
    }

    const inserted = await recordAnalyticsClick({
      eventId,
      code: urlRecord.code,
      userId: urlRecord.userId,
      occurredAt: context.occurredAt,
      referrer: context.referrer,
      browser: context.browser,
      deviceType: context.deviceType,
      country: context.country,
      clientIp: context.clientIp,
      requestId: payload.requestId
    });

    if (!inserted) {
      requestLogger.debug("analytics.process.duplicate_event_skipped", {
        occurredAt: context.occurredAt
      });
      return;
    }

    await Url.updateOne(
      { _id: urlRecord._id },
      {
        $inc: { clickCount: 1 },
        $max: { lastClickedAt: context.occurredAt }
      }
    );

    requestLogger.debug("analytics.mongo.updated", {
      occurredAt: context.occurredAt
    });
    requestLogger.debug("analytics.process.completed");
  } catch (error) {
    requestLogger.error("analytics.process.failed", {
      error
    });
    throw error;
  }
};

import { randomUUID } from "crypto";
import { Queue, Worker } from "bullmq";
import { env } from "../../../shared/src/config/env.js";
import { createBullMqRedisConnection, isRedisEnabled } from "../../../shared/src/config/redis.js";
import { logger } from "../../../shared/src/utils/logger.js";
import { processClickEvent } from "../jobs/clickEventProcessor.js";

let clickQueue = null;
let clickWorker = null;
let workerReadyLogged = false;
let queueUnavailableLogged = false;
let workerConnectionIssueLogged = false;
let queueConnection = null;
let workerConnection = null;

const analyticsLogger = logger.child({ component: "analytics.queue" });
const INLINE_FALLBACK_ATTEMPTS = 3;

const isTransientWorkerConnectionError = (error) => {
  const message = error?.message || "";
  return (
    ["ECONNRESET", "ECONNREFUSED", "ETIMEDOUT", "ENOTFOUND", "EAI_AGAIN"].includes(error?.code)
    || /ECONNRESET|ECONNREFUSED|ETIMEDOUT|ENOTFOUND|Connection is closed|socket closed unexpectedly|Stream isn't writeable/i.test(message)
  );
};

const formatWorkerError = (error) => {
  if (!error) {
    return "Unknown worker error";
  }

  return error.code ? `${error.code}: ${error.message}` : error.message;
};

const normalizeAnalyticsPayload = (payload) => ({
  ...payload,
  eventId: payload?.eventId || randomUUID()
});

const delay = (delayMs) => new Promise((resolve) => {
  setTimeout(resolve, delayMs);
});

const closeRedisClient = async (client) => {
  if (!client) {
    return;
  }

  try {
    await client.quit();
  } catch (_error) {
    try {
      client.disconnect(false);
    } catch (_disconnectError) {
      // no-op
    }
  }
};

const ensureClickQueue = () => {
  if (clickQueue) {
    return clickQueue;
  }

  if (!isRedisEnabled()) {
    if (!queueUnavailableLogged) {
      analyticsLogger.warn("analytics.queue.redis_unavailable");
      queueUnavailableLogged = true;
    }
    return null;
  }

  try {
    queueConnection = createBullMqRedisConnection();
    clickQueue = new Queue("click-events", {
      connection: queueConnection,
      defaultJobOptions: {
        removeOnComplete: 100,
        removeOnFail: 100,
        attempts: 3,
        backoff: {
          type: "exponential",
          delay: 1000
        }
      }
    });

    analyticsLogger.info("analytics.queue.initialized", {
      queueName: "click-events"
    });
    queueUnavailableLogged = false;

    return clickQueue;
  } catch (error) {
    void closeRedisClient(queueConnection);
    queueConnection = null;
    analyticsLogger.warn("analytics.queue.bootstrap_failed", {
      error: formatWorkerError(error)
    });
    return null;
  }
};

const runInlineFallback = (analyticsPayload, requestLogger, reason) => {
  setImmediate(async () => {
    for (let attempt = 1; attempt <= INLINE_FALLBACK_ATTEMPTS; attempt += 1) {
      try {
        await processClickEvent(analyticsPayload);
        requestLogger.debug("analytics.inline.succeeded", {
          reason,
          attempt
        });
        return;
      } catch (error) {
        requestLogger.error("analytics.inline.failed", {
          reason,
          attempt,
          error: formatWorkerError(error)
        });

        if (attempt < INLINE_FALLBACK_ATTEMPTS) {
          await delay(attempt * 1000);
        }
      }
    }

    requestLogger.error("analytics.inline.permanently_failed", {
      reason,
      attempts: INLINE_FALLBACK_ATTEMPTS
    });
  });
};

export const enqueueClickEvent = async (payload) => {
  const analyticsPayload = normalizeAnalyticsPayload(payload);
  const requestLogger = analyticsLogger.child({
    eventId: analyticsPayload.eventId,
    code: analyticsPayload.code,
    requestId: analyticsPayload.requestId
  });
  const queue = ensureClickQueue();

  if (!queue) {
    requestLogger.warn("analytics.queue.unavailable_inline_fallback");
    runInlineFallback(analyticsPayload, requestLogger, "queue_unavailable");
    return;
  }

  try {
    await queue.add("track-click", analyticsPayload, {
      jobId: analyticsPayload.eventId
    });
    requestLogger.debug("analytics.queue.add_succeeded");
  } catch (error) {
    requestLogger.warn("analytics.queue.add_failed_inline_fallback", {
      error: formatWorkerError(error)
    });
    runInlineFallback(analyticsPayload, requestLogger, "queue_add_failed");
  }
};

export const startAnalyticsWorker = () => {
  const queue = ensureClickQueue();

  if (!queue || clickWorker) {
    analyticsLogger.debug("analytics.worker.start_skipped", {
      hasQueue: Boolean(queue),
      alreadyStarted: Boolean(clickWorker)
    });
    return;
  }

  try {
    workerConnection = createBullMqRedisConnection();
    clickWorker = new Worker(
      "click-events",
      async (job) => {
        const jobLogger = analyticsLogger.child({
          jobId: job.id,
          eventId: job.data?.eventId,
          code: job.data?.code,
          requestId: job.data?.requestId
        });
        const lagMs = Math.max(Date.now() - Number(job.timestamp || Date.now()), 0);

        jobLogger.debug("analytics.job.started", {
          lagMs
        });
        await processClickEvent(job.data);
        jobLogger.debug("analytics.job.completed");
      },
      {
        connection: workerConnection,
        concurrency: env.analyticsWorkerConcurrency
      }
    );
  } catch (error) {
    void closeRedisClient(workerConnection);
    workerConnection = null;
    clickWorker = null;
    analyticsLogger.warn("analytics.worker.bootstrap_failed", {
      error: formatWorkerError(error)
    });
    return;
  }

  clickWorker.on("ready", () => {
    workerConnectionIssueLogged = false;

    if (workerReadyLogged) {
      analyticsLogger.info("analytics.worker.reconnected");
      return;
    }

    workerReadyLogged = true;
    analyticsLogger.info("analytics.worker.ready", {
      concurrency: env.analyticsWorkerConcurrency
    });
  });

  clickWorker.on("failed", (job, error) => {
    analyticsLogger.error("analytics.job.failed", {
      jobId: job?.id || "unknown",
      eventId: job?.data?.eventId,
      code: job?.data?.code,
      requestId: job?.data?.requestId,
      error: formatWorkerError(error)
    });
  });

  clickWorker.on("error", (error) => {
    if (isTransientWorkerConnectionError(error)) {
      if (!workerConnectionIssueLogged) {
        analyticsLogger.warn("analytics.worker.redis_connection_reset", {
          error: formatWorkerError(error)
        });
        workerConnectionIssueLogged = true;
      }
      return;
    }

    analyticsLogger.error("analytics.worker.error", {
      error: formatWorkerError(error)
    });
  });
};

export const stopAnalyticsWorker = async () => {
  const closingTasks = [];

  if (clickWorker) {
    closingTasks.push(clickWorker.close());
    clickWorker = null;
  }

  if (clickQueue) {
    closingTasks.push(clickQueue.close());
    clickQueue = null;
  }

  if (workerConnection) {
    closingTasks.push(closeRedisClient(workerConnection));
    workerConnection = null;
  }

  if (queueConnection) {
    closingTasks.push(closeRedisClient(queueConnection));
    queueConnection = null;
  }

  workerReadyLogged = false;
  queueUnavailableLogged = false;
  workerConnectionIssueLogged = false;

  await Promise.allSettled(closingTasks);
};

        "user-agent": "PKLinks/1.0"
      }
    },
    (response) => {
      let body = "";

      response.setEncoding("utf8");
      response.on("data", (chunk) => {
        body += chunk;
      });
      response.on("end", () => {
        const responseBody = String(body || "").trim();

        if (!responseBody) {
          resolve({
            ok: response.statusCode >= 200 && response.statusCode < 300,
            status: response.statusCode || 0,
            payload: null
          });
          return;
        }

        try {
          resolve({
            ok: response.statusCode >= 200 && response.statusCode < 300,
            status: response.statusCode || 0,
            payload: JSON.parse(responseBody)
          });
        } catch (error) {
          reject(new Error(`Invalid JSON response from country lookup service: ${error.message}`));
        }
      });
    }
  );

  request.setTimeout(timeoutMs, () => {
    request.destroy(new Error("Lookup timed out"));
  });

  request.on("error", reject);
  request.end();
});

const normalizeClientIp = (value) => {
  const rawValue = String(value || "").trim();

  if (!rawValue) {
    return "";
  }

  const firstValue = rawValue.split(",")[0].trim();
  const withoutIpv6Prefix = firstValue.replace(/^::ffff:/i, "");

  if (/^\d{1,3}(?:\.\d{1,3}){3}:\d+$/.test(withoutIpv6Prefix)) {
    return withoutIpv6Prefix.replace(/:\d+$/, "");
  }

  return withoutIpv6Prefix;
};

const isPrivateIpv4 = (ip) => {
  if (!/^\d{1,3}(?:\.\d{1,3}){3}$/.test(ip)) {
    return false;
  }

  const parts = ip.split(".").map(Number);
  const [first, second] = parts;

  return (
    first === 10
    || first === 127
    || first === 0
    || (first === 192 && second === 168)
    || (first === 172 && second >= 16 && second <= 31)
    || (first === 169 && second === 254)
  );
};

const isPrivateIp = (ip) => {
  const normalized = normalizeClientIp(ip).toLowerCase();

  if (!normalized || normalized === "unknown" || normalized === "localhost" || normalized === "::1") {
    return true;
  }

  if (isPrivateIpv4(normalized)) {
    return true;
  }

  return (
    normalized.startsWith("fc")
    || normalized.startsWith("fd")
    || normalized.startsWith("fe80:")
  );
};

const isCountryLookupCircuitOpen = () => (
  countryLookupCircuitState.nextAttemptAt > 0
  && Date.now() < countryLookupCircuitState.nextAttemptAt
);

const resetCountryLookupCircuit = () => {
  countryLookupCircuitState.consecutiveFailures = 0;
  countryLookupCircuitState.openedAt = 0;
  countryLookupCircuitState.nextAttemptAt = 0;
};

const recordCountryLookupFailure = (requestLogger, details = {}) => {
  countryLookupCircuitState.consecutiveFailures += 1;

  if (countryLookupCircuitState.consecutiveFailures < env.countryLookupCircuitBreakerFailures) {
    return;
  }

  countryLookupCircuitState.openedAt = Date.now();
  countryLookupCircuitState.nextAttemptAt = countryLookupCircuitState.openedAt + env.countryLookupCircuitBreakerResetMs;

  requestLogger.warn("country.lookup.circuit_opened", {
    consecutiveFailures: countryLookupCircuitState.consecutiveFailures,
    resetAfterMs: env.countryLookupCircuitBreakerResetMs,
    ...details
  });
};

const recordCountryLookupSuccess = () => {
  if (
    countryLookupCircuitState.consecutiveFailures > 0
    || countryLookupCircuitState.nextAttemptAt > 0
    || countryLookupCircuitState.openedAt > 0
  ) {
    resetCountryLookupCircuit();
  }
};

const fetchCountryFromLookupService = async (ip, context = {}) => {
  const requestLogger = countryLogger.child({ ip, ...context });
  const lookupUrl = env.countryLookupUrlTemplate.replace("{ip}", encodeURIComponent(ip));

  if (isCountryLookupCircuitOpen()) {
    requestLogger.warn("country.lookup.circuit_open", {
      consecutiveFailures: countryLookupCircuitState.consecutiveFailures,
      retryInMs: Math.max(countryLookupCircuitState.nextAttemptAt - Date.now(), 0)
    });
    return UNKNOWN_COUNTRY;
  }

  try {
    const response = await requestJson(lookupUrl, env.countryLookupTimeoutMs);

    if (!response.ok) {
      requestLogger.warn("country.lookup.failed", {
        statusCode: response.status,
        response: response.payload
      });
      recordCountryLookupFailure(requestLogger, {
        statusCode: response.status
      });
      return UNKNOWN_COUNTRY;
    }

    const payload = response.payload;
    const country = normalizeCountryLabel(
      payload?.country
      || payload?.country_name
      || payload?.countryName
      || payload?.country_code
      || payload?.countryCode
    );

    if (country === UNKNOWN_COUNTRY) {
      requestLogger.debug("country.lookup.empty");
      return UNKNOWN_COUNTRY;
    }

    requestLogger.debug("country.lookup.success", {
      country
    });
    recordCountryLookupSuccess();
    return country;
  } catch (error) {
    requestLogger.warn("country.lookup.failed", {
      error: error?.message || error
    });
    recordCountryLookupFailure(requestLogger, {
      error: error?.message || error
    });
    return UNKNOWN_COUNTRY;
  }
};

export const resolveCountry = async ({ country, clientIp, ...context } = {}) => {
  const normalizedCountry = normalizeCountryLabel(country);

  if (normalizedCountry !== UNKNOWN_COUNTRY) {
    return normalizedCountry;
  }

  if (!env.countryLookupEnabled) {
    return UNKNOWN_COUNTRY;
  }

  const normalizedIp = normalizeClientIp(clientIp);

  if (!normalizedIp || (!env.countryLookupAllowPrivate && isPrivateIp(normalizedIp))) {
    return UNKNOWN_COUNTRY;
  }

  const cachedCountry = normalizeCountryLabel(await readCountryFromCache(normalizedIp));

  if (cachedCountry !== UNKNOWN_COUNTRY) {
    countryLogger.debug("country.cache.hit", {
      ip: normalizedIp,
      country: cachedCountry,
      ...context
    });
    return cachedCountry;
  }

  const lookedUpCountry = await fetchCountryFromLookupService(normalizedIp, context);

  if (lookedUpCountry !== UNKNOWN_COUNTRY) {
    await writeCountryToCache(normalizedIp, lookedUpCountry);
  }

  return lookedUpCountry;
};

const HOP_BY_HOP_HEADERS = new Set([
  "connection",
  "content-length",
  "host",
  "keep-alive",
  "proxy-authenticate",
  "proxy-authorization",
  "te",
  "trailer",

const getRequestBody = (req) => {
  if (req.method === "GET" || req.method === "HEAD") {
    return undefined;
  }

  if (req.body === undefined || req.body === null) {
    return undefined;
  }

  return JSON.stringify(req.body);
};

const copyResponseHeaders = (sourceHeaders, res) => {
  const setCookies = typeof sourceHeaders.getSetCookie === "function"
    ? sourceHeaders.getSetCookie()
    : [];

  if (setCookies.length > 0) {
    res.setHeader("set-cookie", setCookies);
  }

  sourceHeaders.forEach((value, key) => {
    if (
      key.toLowerCase() !== "set-cookie"
      && !HOP_BY_HOP_HEADERS.has(key.toLowerCase())
    ) {
      res.setHeader(key, value);
    }
  });
};

const sendGenericUpstreamError = (res) => {
  res.removeHeader("content-length");
  res.setHeader("content-type", "application/json; charset=utf-8");
  res.send(JSON.stringify({
    success: false,
    message: "Internal server error.",
    details: null
  }));
};

const buildForwardHeaders = (req, hasBody) => {
  const headers = new Headers();
  const forwardedFor = req.get("x-forwarded-for");

  if (req.get("accept")) {
    headers.set("accept", req.get("accept"));
  }

  if (req.get("authorization")) {
    headers.set("authorization", req.get("authorization"));
  }

  if (req.get("cookie")) {
    headers.set("cookie", req.get("cookie"));
  }

  if (req.get("user-agent")) {
    headers.set("user-agent", req.get("user-agent"));
  }

  if (req.requestId) {
    headers.set("x-request-id", req.requestId);
  }

  headers.set(
    "x-forwarded-for",
    forwardedFor ? `${forwardedFor}, ${req.ip || "unknown"}` : req.ip || "unknown"
  );
  headers.set("x-forwarded-host", req.get("host") || "");
  headers.set("x-forwarded-proto", req.protocol || "http");

  if (hasBody) {
    headers.set("content-type", "application/json");
  }

  return headers;
};

export const forwardRequest = async ({
  req,
  res,
  targetBaseUrl,
  targetPath = req.path,
  search = req.originalUrl.includes("?") ? req.originalUrl.slice(req.originalUrl.indexOf("?")) : ""
}) => {
  const targetUrl = new URL(targetPath, targetBaseUrl);
  const searchParams = new URLSearchParams(search.startsWith("?") ? search.slice(1) : search);
  targetUrl.search = searchParams.toString();

  const body = getRequestBody(req);
  const response = await fetch(targetUrl, {
    method: req.method,
    headers: buildForwardHeaders(req, Boolean(body)),
    body,
    redirect: "manual"
  });

  copyResponseHeaders(response.headers, res);
  res.status(response.status);

  if (req.method === "HEAD" || response.status === 204 || response.status === 304) {
    res.end();
    return;
  }

  const responseText = await response.text();

  if (response.status >= 500) {
    req.log?.error("gateway.upstream.server_error", {
      targetUrl: targetUrl.toString(),
      statusCode: response.status,
      responseBody: responseText.slice(0, 1000)
    });
    sendGenericUpstreamError(res);
    return;
  }

  res.send(responseText);
};

const parseDurationToMs = (value, fallbackMs) => {
  const normalized = String(value || "").trim();

  if (!normalized) {
    return fallbackMs;
  }

  if (/^\d+$/.test(normalized)) {
    return Number(normalized);
  }

  const match = normalized.match(/^(\d+)(ms|s|m|h|d)$/i);

  if (!match) {
    return fallbackMs;
  }

  const amount = Number(match[1]);
  const unit = match[2].toLowerCase();
  const multipliers = {
    ms: 1,
    s: 1000,
    m: 60 * 1000,
    h: 60 * 60 * 1000,
    d: 24 * 60 * 60 * 1000
  };

  return amount * multipliers[unit];
};

import { Url } from "../models/Url.js";
import { AppError } from "../utils/AppError.js";
import { isReservedCode, sanitizeAlias } from "../utils/base62.js";

const ALIAS_PATTERN = /^[a-zA-Z0-9][a-zA-Z0-9-_]{2,39}$/;
const SHORT_ID_LENGTH = 6;
const SHORT_ID_ALPHABET = "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ";
const MAX_GENERATION_ATTEMPTS = 10;

const generateRandomCode = (length = SHORT_ID_LENGTH) => {
  let code = "";

  for (let index = 0; index < length; index += 1) {
    code += SHORT_ID_ALPHABET[randomInt(SHORT_ID_ALPHABET.length)];
  }

  return code;
};

export const validateCustomAlias = async (value) => {
  const alias = sanitizeAlias(value);

  if (!ALIAS_PATTERN.test(alias) || isReservedCode(alias)) {
    throw new AppError(
      "Custom alias must be 3-40 characters and use only letters, numbers, hyphens, or underscores.",
      400
    );
  }

  const existing = await Url.exists({ code: alias });

  if (existing) {
    throw new AppError("That custom alias is already taken.", 409);
  }

  return alias;
};

export const generateUniqueCode = async () => {
  for (let attempt = 0; attempt < MAX_GENERATION_ATTEMPTS; attempt += 1) {
    const candidate = generateRandomCode();

    if (isReservedCode(candidate)) {
      continue;
    }

    const existing = await Url.exists({ code: candidate });

    if (!existing) {
      return candidate;
    }
  }

  throw new AppError("Unable to generate a unique short code. Please try again.", 503);
};

  }

  if (/^\d{1,3}(?:\.\d{1,3}){3}:\d+$/.test(withoutIpv6Prefix)) {

    return false;

  return (
    first === 10
    || first === 127
    || first === 0
    || (first === 192 && second === 168)
    || (first === 172 && second >= 16 && second <= 31)
    || (first === 169 && second === 254)

  });
};

  const normalized = String(value || "").trim();

  if (!normalized) {

  if (/^\d+$/.test(normalized)) {
    return Number(normalized);
  }

  const match = normalized.match(/^(\d+)(ms|s|m|h|d)$/i);

  if (!match) {
    return fallbackMs;
  }

  const amount = Number(match[1]);
  const unit = match[2].toLowerCase();
  const multipliers = {
    ms: 1,
    s: 1000,
    m: 60 * 1000,
    h: 60 * 60 * 1000,
    d: 24 * 60 * 60 * 1000

};

    if (currentSession) {
      await AuthSession.deleteOne({ _id: currentSession._id });

    throw new AppError("Authentication is required.", 401);

  if (!user) {
    await AuthSession.deleteOne({ _id: currentSession._id });
    throw new AppError("Authentication is required.", 401);
  }

  currentSession.lastUsedAt = new Date();
  await currentSession.save();

  return buildSessionPayload(user, req, currentSession);
};

  const normalizedName = normalizeName(name);

  if (!uid) {
    throw new AppError("Authentication is required.", 401);
  }

  let user = await User.findOne({ uid });

  if (!user && normalizedEmail) {
    user = await User.findOne({ email: normalizedEmail });
  }

  if (!user) {
    user = await User.create({
      uid,
      email: normalizedEmail,
      name: normalizedName,
      authProvider: "google"
    });
  } else {
    if (user.authProvider === "email" && user.uid !== uid) {
      throw new AppError("Use email sign-in for this account.", 409);
    }

    user.uid = uid;
    user.email = normalizedEmail || user.email;
    user.name = normalizedName || user.name;
    user.authProvider = "google";
    await user.save();
  }

  return user;
};

  stateToken,

  req

  }

  if (decodedState?.type !== "google_oauth_state" || hashValue(decodedState.nonce) !== stateNonceHash) {
    throw new AppError("Google sign-in could not be completed.", 401);

  if (!googleTokenData?.access_token) {
    throw new AppError("Google sign-in could not be completed.", 401);

    uid: googleProfile.sub,
    email: googleProfile.email,
    name: googleProfile.name
  });

  return buildSessionPayload(user, req);
};

const CACHE_PREFIX = "pklinks:code:";
const NEGATIVE_CACHE_VALUE = "__missing__";
const cacheLogger = logger.child({ component: "cache" });

const getCacheKey = (code) => `${CACHE_PREFIX}${code}`;

const getTTLFromExpiry = (expiry) => {
  if (!expiry) {
    return 60 * 60 * 24 * 7;
  }

  const ttlSeconds = Math.floor((new Date(expiry).getTime() - Date.now()) / 1000);
  return Math.max(ttlSeconds, 60);
};

export const readUrlFromCache = async (code, context = {}) => {
  const requestLogger = cacheLogger.child({ code, ...context });

  if (!redis) {
    requestLogger.debug("cache.read.redis_unavailable");
    return { status: "miss", value: null };
  }

  try {
    const rawValue = await redis.get(getCacheKey(code));

    if (!rawValue) {
      requestLogger.debug("cache.read.miss");
      return { status: "miss", value: null };
    }

    if (rawValue === NEGATIVE_CACHE_VALUE) {
      requestLogger.debug("cache.read.negative_hit");
      return { status: "negative", value: null };
    }

    requestLogger.debug("cache.read.hit");

    return { status: "hit", value: JSON.parse(rawValue) };
  } catch (error) {
    requestLogger.warn("cache.read.failed", {
      error
    });
    return { status: "miss", value: null };
  }
};

export const writeUrlToCache = async (urlRecord, context = {}) => {
  const ttlSeconds = getTTLFromExpiry(urlRecord.expiry);
  const requestLogger = cacheLogger.child({ code: urlRecord.code, ...context });

  if (!redis) {
    requestLogger.debug("cache.write.redis_unavailable", {
      ttlSeconds
    });
    return;
  }

  try {
    await redis.set(
      getCacheKey(urlRecord.code),
      JSON.stringify({
        code: urlRecord.code,
        longUrl: urlRecord.longUrl,
        expiry: urlRecord.expiry,
        isActive: urlRecord.isActive
      }),
      "EX",
      ttlSeconds
    );

    requestLogger.debug("cache.write.success", {
      ttlSeconds,
      isActive: urlRecord.isActive
    });
  } catch (error) {
    requestLogger.warn("cache.write.failed", {
      ttlSeconds,
      error
    });
  }
};

import { randomUUID } from "crypto";
import morgan from "morgan";

const LEVELS = new Set(["debug", "info", "warn", "error"]);

const serialize = (value) => {
  if (value instanceof Error) {
    return {
      name: value.name,
      message: value.message,
      stack: value.stack
    };
  }

  if (value instanceof Date) {
    return value.toISOString();
  }

  if (Array.isArray(value)) {
    return value.map(serialize);
  }

  if (value && typeof value === "object") {
    return Object.fromEntries(
      Object.entries(value)
        .filter(([, entryValue]) => entryValue !== undefined)
        .map(([key, entryValue]) => [key, serialize(entryValue)])
    );
  }

  return value;
};

const emit = (level, message, meta = {}) => {
  const entry = {
    timestamp: new Date().toISOString(),
    level,
    message,
    ...serialize(meta)
  };

  const output = level === "error" || level === "warn" ? console.error : console.log;
  output(JSON.stringify(entry));
};

const buildLogger = (baseMeta = {}) => {
  const childLogger = {
    debug: (message, meta) => emit("debug", message, { ...baseMeta, ...meta }),
    info: (message, meta) => emit("info", message, { ...baseMeta, ...meta }),
    warn: (message, meta) => emit("warn", message, { ...baseMeta, ...meta }),
    error: (message, meta) => emit("error", message, { ...baseMeta, ...meta }),
    child: (childMeta = {}) => buildLogger({ ...baseMeta, ...childMeta }),
    log: ({ level = "info", message = "log", ...meta } = {}) => {
      const normalizedLevel = LEVELS.has(level) ? level : "info";
      emit(normalizedLevel, message, { ...baseMeta, ...meta });
    }
  };

  return childLogger;
};
